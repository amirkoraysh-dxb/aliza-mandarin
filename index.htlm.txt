    <button class="mode-btn" id="btnModeListen" onclick="setMode('listen')">🔊 Listen & Find</button>

  <div id="winBanner" class="win-banner">🎉 太棒了! (Well Done, Aliza!) 🎉</div>

  <script>
    // --- LESSON DATA REPOSITORY ---
    // (This list gets expanded whenever you share new lesson PDFs)
    const LESSONS = {
      animals: {
        name: "🐶 Animals (动物)",
        words: [
          { hanzi: "狗", pinyin: "gǒu", en: "Dog", icon: "🐶" },
          { hanzi: "猫", pinyin: "māo", en: "Cat", icon: "🐱" },
          { hanzi: "兔子", pinyin: "tù zi", en: "Rabbit", icon: "🐰" },
          { hanzi: "鸟", pinyin: "niǎo", en: "Bird", icon: "🐦" }
        ]
      },
      fruits: {
        name: "🍎 Fruits (水果)",
        words: [
          { hanzi: "苹果", pinyin: "píng guǒ", en: "Apple", icon: "🍎" },
          { hanzi: "香蕉", pinyin: "xiāng jiāo", en: "Banana", icon: "🍌" },
          { hanzi: "西瓜", pinyin: "xī guā", en: "Watermelon", icon: "🍉" },
          { hanzi: "草莓", pinyin: "cǎo méi", en: "Strawberry", icon: "🍓" }
        ]
      },
      colors: {
        name: "🎨 Colors (颜色)",
        words: [
          { hanzi: "红", pinyin: "hóng", en: "Red", icon: "🔴" },
          { hanzi: "蓝", pinyin: "lán", en: "Blue", icon: "🔵" },
          { hanzi: "黄", pinyin: "huáng", en: "Yellow", icon: "🟡" },
          { hanzi: "绿", pinyin: "lǜ", en: "Green", icon: "🟢" }
        ]
      },
      numbers: {
        name: "🔢 Numbers (数字)",
        words: [
          { hanzi: "一", pinyin: "yī", en: "One (1)", icon: "1️⃣" },
          { hanzi: "二", pinyin: "èr", en: "Two (2)", icon: "2️⃣" },
          { hanzi: "三", pinyin: "sān", en: "Three (3)", icon: "3️⃣" },
          { hanzi: "四", pinyin: "sì", en: "Four (4)", icon: "4️⃣" }
        ]
      }
    };

    let currentTopicKey = 'animals';
    let currentMode = 'memory'; // 'memory' or 'listen'
    let flippedCards = [];
    let lockBoard = false;
    let targetWord = null;

    // Speech synthesis for iPad
    function speak(text) {
      if ('speechSynthesis' in window) {
        window.speechSynthesis.cancel();
        const utterance = new SpeechSynthesisUtterance(text);
        utterance.lang = 'zh-CN';
        utterance.rate = 0.8; // slightly slower for a 4-year-old
        window.speechSynthesis.speak(utterance);
      }
    }

    function initTopicBar() {
      const bar = document.getElementById('topicBar');
      bar.innerHTML = '';
      Object.keys(LESSONS).forEach(key => {
        const btn = document.createElement('button');
        btn.className = `topic-btn ${key === currentTopicKey ? 'active' : ''}`;
        btn.innerText = LESSONS[key].name;
        btn.onclick = () => {
          currentTopicKey = key;
          initTopicBar();
          loadGame();
        };
        bar.appendChild(btn);
      });
    }

    function setMode(mode) {
      currentMode = mode;
      document.getElementById('btnModeMemory').classList.toggle('active', mode === 'memory');
      document.getElementById('btnModeListen').classList.toggle('active', mode === 'listen');
      document.getElementById('audio-prompt').style.display = mode === 'listen' ? 'block' : 'none';
      loadGame();
    }

    function shuffle(array) {
      return array.sort(() => Math.random() - 0.5);
    }

    function loadGame() {
      const grid = document.getElementById('grid');
      const winBanner = document.getElementById('winBanner');
      winBanner.style.display = 'none';
      grid.innerHTML = '';
      flippedCards = [];
      lockBoard = false;

      const items = LESSONS[currentTopicKey].words;

      if (currentMode === 'memory') {
        // Pairs: 1 Chinese card + 1 English/Icon card
        let cards = [];
        items.forEach((item, index) => {
          cards.push({ id: index, type: 'zh', text: item.hanzi, pinyin: item.pinyin, speak: item.hanzi });
          cards.push({ id: index, type: 'en', text: item.en, icon: item.icon, speak: item.hanzi });
        });
        cards = shuffle(cards);
        grid.className = 'grid-4x2';

        cards.forEach(cardData => {
          const card = document.createElement('div');
          card.className = 'card hidden-face';
          card.dataset.id = cardData.id;

          card.onclick = () => {
            if (lockBoard || card.classList.contains('matched') || card === flippedCards[0]?.el) return;

            // Reveal
            card.classList.remove('hidden-face');
            if (cardData.type === 'zh') {
              card.innerHTML = `<div class="card-hanzi">${cardData.text}</div><div class="card-pinyin">${cardData.pinyin}</div>`;
            } else {
              card.innerHTML = `<div class="card-icon">${cardData.icon}</div><div class="card-english">${cardData.text}</div>`;
            }
            speak(cardData.speak);

            flippedCards.push({ el: card, data: cardData });

            if (flippedCards.length === 2) {
              lockBoard = true;
              const [first, second] = flippedCards;
              if (first.data.id === second.data.id) {
                // Match
                setTimeout(() => {
                  first.el.classList.add('matched');
                  second.el.classList.add('matched');
                  flippedCards = [];
                  lockBoard = false;
                  checkWin();
                }, 400);
              } else {
                // Not a match
                setTimeout(() => {
                  first.el.classList.add('hidden-face');
                  second.el.classList.add('hidden-face');
                  first.el.innerHTML = '';
                  second.el.innerHTML = '';
                  flippedCards = [];
                  lockBoard = false;
                }, 900);
              }
            }
          };
          grid.appendChild(card);
        });

      } else {
        // Listen & Match Mode
        grid.className = 'grid-4x2';
        targetWord = items[Math.floor(Math.random() * items.length)];
        setTimeout(() => speak(targetWord.hanzi), 300);

        shuffle([...items]).forEach(item => {
          const card = document.createElement('div');
          card.className = 'card';
          card.innerHTML = `
            <div class="card-icon">${item.icon}</div>
            <div class="card-hanzi">${item.hanzi}</div>
            <div class="card-english">${item.en}</div>
          `;

          card.onclick = () => {
            if (item.hanzi === targetWord.hanzi) {
              card.classList.add('matched');
              speak("对了! " + item.hanzi); // "Correct!"
              setTimeout(() => {
                winBanner.style.display = 'block';
                setTimeout(loadGame, 1400);
              }, 400);
            } else {
              speak("不正确");
              card.style.borderColor = '#FF4D4F';
              setTimeout(() => card.style.borderColor = '#F0ECE4', 500);
            }
          };
          grid.appendChild(card);
        });
      }
    }

    function playCurrentPrompt() {
      if (targetWord) speak(targetWord.hanzi);
    }

    function checkWin() {
      const allMatched = document.querySelectorAll('.card.matched').length;
      const total = LESSONS[currentTopicKey].words.length * 2;
      if (allMatched === total) {
        document.getElementById('winBanner').style.display = 'block';
      }
    }

    // Initialize
    initTopicBar();
    loadGame();
  </script>
